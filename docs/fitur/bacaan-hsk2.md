<script setup>
import { onMounted } from 'vue'
onMounted(() => {
  var sheet = document.getElementById('dk-sheet');
  if (!sheet) return;
  var trOn = true, pyOn = true;
  var T = { tr: "Terjemahan", py: "Pinyin" };

  function speak(text) {
    try {
      var u = new SpeechSynthesisUtterance(text);
      u.lang = 'zh-CN'; u.rate = 0.8;
      speechSynthesis.cancel(); speechSynthesis.speak(u);
    } catch (e) {}
  }

  document.querySelectorAll('.dk-word').forEach(function(b) {
    b.addEventListener('click', function() {
      document.getElementById('dk-w').textContent = b.dataset.w;
      document.getElementById('dk-p').textContent = b.dataset.p;
      document.getElementById('dk-m').textContent = b.dataset.m;
      document.getElementById('dk-hsk').textContent = 'HSK ' + b.dataset.h;
      document.getElementById('dk-say').onclick = function() { speak(b.dataset.w); };
      sheet.hidden = false;
    });
  });
  document.querySelector('.dk-x').addEventListener('click', function() { sheet.hidden = true; });
  sheet.addEventListener('click', function(e) { if (e.target === sheet) sheet.hidden = true; });

  document.querySelectorAll('.dk-say-sent').forEach(function(b) {
    b.addEventListener('click', function() { speak(b.dataset.sent); });
  });

  var trBtn = document.getElementById('dk-tr');
  trBtn.addEventListener('click', function() {
    trOn = !trOn;
    trBtn.classList.toggle('on', trOn);
    trBtn.textContent = T.tr + ' ' + (trOn ? 'ON' : 'OFF');
    document.getElementById('dk-passage').classList.toggle('hide-tr', !trOn);
  });
  trBtn.textContent = T.tr + ' ON';
  var pyBtn = document.getElementById('dk-py');
  pyBtn.addEventListener('click', function() {
    pyOn = !pyOn;
    pyBtn.classList.toggle('on', pyOn);
    pyBtn.textContent = T.py + ' ' + (pyOn ? 'ON' : 'OFF');
    document.getElementById('dk-passage').classList.toggle('hide-py', !pyOn);
  });
  pyBtn.textContent = T.py + ' ON';
})
</script>
